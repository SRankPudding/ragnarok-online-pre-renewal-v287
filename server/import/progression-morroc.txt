// The two harbor assignments are independent of the other city chapters.
function	script	F_communityHarbor	{
	.@quest = getarg(0);
	.@state = checkquest(.@quest);
	mes "[" + getarg(1) + "]";
	if (.@state < 0) {
		if (BaseClass == Job_Novice || BaseLevel < 10) {
			mes "Return after your first job change and Base Level 10.";
			close;
		}
		mes getarg(2);
		mes "The first chamber contains stronger monsters than the fields. Bring healing items, fight one enemy at a time, and consider a shield if your class can use one.";
		mes "For gear and refining advice, visit the Equipment Mentor in Prontera.";
		if (select("Accept the assignment:Later") != 1) close;
		setquest .@quest;
	}
	if (.@state == 2) {
		mes "Your report is already in the harbor ledger.";
		close;
	}
	if (checkquest(.@quest,HUNTING) != 2) {
		mes getarg(3);
		close;
	}
	if (!checkweight(504,10)) {
		mes "Make room for the ten White Potions in your reward.";
		close;
	}
	completequest .@quest;
	callfunc "F_communityQuestReward",550,650;
	getitem 504,10;
	mes "Supplies received: 10 White Potions.";
	mes "I will send your findings to the guild. Thank you for keeping the sea route open.";
	close;
}

function	script	F_communityLevelCheckpoint	{
	.@base = getarg(0);
	.@job = getarg(1);
	.@job = (Class >= Job_Swordman && Class <= Job_Thief) ? .@job : 0;
	if (BaseLevel >= .@base && JobLevel >= .@job) return;
	sroprogressfloor .@base,.@job;
	mes "Training checkpoint reached: Base Level " + BaseLevel + ", Job Level " + JobLevel + ".";
	return;
}
