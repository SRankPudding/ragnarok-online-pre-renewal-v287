prontera,156,182,4	script	City Crystal#prontera	1914,{
	mes "[City Crystal]";
	mes "Travel to another City Crystal for 100 Zeny.";
	next;
	.@choice = select("Prontera (Acolyte):Izlude (Swordman):Morroc (Thief):Geffen (Mage):Payon:Archer Village (Archer):Alberta (Merchant):Al De Baran:Amatsu:Ayothaya:Brasilis:Comodo:Einbech:Einbroch:Gonryun:Lighthalzen:Louyang:Moscovia:Niflheim:Umbala:Yuno:Cancel");
	if (.@choice == 22) close;
	setarray .@map$[0], "prontera","izlude","morocc","geffen","payon","pay_arche","alberta","aldebaran","amatsu","ayothaya","brasilis","comodo","einbech","einbroch","gonryun","lighthalzen","louyang","moscovia","niflheim","umbala","yuno";
	setarray .@x[0], 156,127,156,119,178,56,117,146,104,214,199,194,183,244,161,166,212,225,203,86,154;
	setarray .@y[0], 182,116,94,59,104,123,60,119,149,169,221,150,132,205,122,100,104,191,179,160,187;
	.@i = .@choice - 1;
	if (strcharinfo(3) == .@map$[.@i]) {
		mes "You are already in that city.";
		close;
	}
	if (Zeny < 100) {
		mes "You need 100 Zeny to travel.";
		close;
	}
	.@landing = .@y[.@i] + 2;
	if (!checkcell(.@map$[.@i], .@x[.@i], .@landing, cell_chkpass)) {
		mes "This crystal is currently unavailable.";
		close;
	}
	Zeny -= 100;
	close2;
	warp .@map$[.@i], .@x[.@i], .@landing;
	end;
}
